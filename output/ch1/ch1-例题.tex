\begin{enumerate}[label=\textbf{例\arabic*.}, leftmargin=*]

% 例 1
\item 一小船被一根跨过小滑轮的绳子牵引，绳端以恒定速率 $v_0$ 收缩。设滑轮距水面高度为 $h$，开始时绳长 $l_0$。求小船运动方程 $x(t)$。

% 例 2
\item 一质点沿平面曲线运动，已知 $\vec{r}(t) = 2t\,\vec{i} + (2 - t^2)\,\vec{j}$（单位：m，$t$ 单位：s）。
\begin{enumerate}[label=(\arabic*)]
  \item 求 $t = 1\,\mathrm{s}$ 和 $t = 2\,\mathrm{s}$ 时的位矢，并计算这一秒内的位移和平均速度；
  \item 求任意时刻的速度和加速度；
  \item 求轨道方程。
\end{enumerate}

% 例 3
\item 已知质点作半径为 $R$ 的匀速圆周运动，角速度为 $\omega$。写出其运动学方程 $\vec{r}(t)$，并求 $\vec{v}$、$\vec{a}$ 的大小。

% 例 4
\item 直杆 $AB$，$A$ 端沿 $y$ 轴、$B$ 端沿 $x$ 轴正方向滑动（$A$ 在 $y$ 轴正方向，$B$ 在 $x$ 轴正方向）。设 $B$ 端以速度 $v$ 匀速运动，初始时杆与 $x$ 轴成 $\varphi_0$ 角。求 $A$ 端的速度与加速度。

% 例 5
\item 课后题 1.4：与例 1 相同的滑轮拉船情形，但求（1）船的速率（$v_{\text{船}}=v_0\cos\theta$，$\theta$ 为绳与水面夹角）；（2）船的加速度；（3）$\theta \to 0$（即船接近滑轮正下方）时船的极限速度。

% 例 6
\item 一质点沿 $x$ 轴运动，加速度为 $a = 100 - 4t^2$（单位：m/s²），初始条件：$t=0$ 时 $v_0=0$，$x_0=0$。求：
\begin{enumerate}[label=(\arabic*)]
  \item 速度 $v(t)$；
  \item 运动学方程 $x(t)$。
\end{enumerate}

% 例 7
\item 斜抛运动：初速度 $v_0$ 与水平方向成 $\theta$ 角。求：
\begin{enumerate}[label=(\arabic*)]
  \item 运动学方程；
  \item 轨迹方程；
  \item 飞行时间；
  \item 水平射程；
  \item 最大高度。
\end{enumerate}

% 例 8
\item 一汽车沿半径 $R = 1000\,\mathrm{m}$ 的圆弧公路行驶，运动学方程 $s = 20t - 0.2t^2$（单位：m，s）。求 $t = 5\,\mathrm{s}$ 时的切向加速度、法向加速度和总加速度。

% 例 9
\item 一质点从原点出发沿 $x$ 轴正方向运动，加速度 $a_x = kx$（$k$ 为常量），初始条件 $v_0 = 0$，$x_0 = 0$。求速度 $v(x)$。

% 例 10
\item 一质点作圆周运动，角速度 $\omega = k t^2$（$k$ 为常量）。设 $t=0$ 时 $\theta=0$，求：
\begin{enumerate}[label=(\arabic*)]
  \item 角加速度 $\beta$；
  \item 角位移 $\theta(t)$；
  \item 切向加速度和法向加速度。
\end{enumerate}

% 例 11
\item 一长为 $l$ 的细杆 $OA$，$O$ 端在水平面上以匀角速度 $\omega_0$ 转动，$A$ 端在竖直平面内的铅垂杆上滑动。已知 $\theta$ 为 $OA$ 与水平面的夹角。
\begin{enumerate}[label=(\arabic*)]
  \item 写出 $A$ 端的位置坐标 $y_A$（以 $O$ 在水平面为原点）；
  \item 求 $A$ 端的速度和加速度大小；
  \item 已知 $\beta = \dfrac{3g}{2l}\cos\theta$，$\theta_0 = 0$，由动力学求杆转动至 $\theta = 60°$ 时 $A$ 端的速度。
\end{enumerate}

% 例 12
\item 一辆卡车以速度 $v_1$ 在水平直路上行驶，雨滴相对地面以速度 $v_2$ 竖直下落。一人站在卡车上用雨伞挡雨。求雨滴相对卡车的速度。

\end{enumerate}
